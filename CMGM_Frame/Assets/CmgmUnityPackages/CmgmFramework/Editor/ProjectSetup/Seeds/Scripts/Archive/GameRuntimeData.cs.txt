using System.Collections.Generic;
using CMGM.Data;

namespace CMGM.Workspace
{
    /// <summary>
    /// 游戏运行时存档数据（仅在 <see cref="CMGM.Data.ArchiveManager"/> 中持有唯一实例）。
    /// </summary>
    public class GameRuntimeData : ISaveable
    {
        /// <summary>业务 payload schema 版本（Migrator / 多 Chunk 前置；当前仍为整对象 JSON）。</summary>
        public int SaveSchemaVersion = 1;
        public WorldSaveData World = new WorldSaveData();
        public QuestSaveData Quest = new QuestSaveData();
        public InventorySaveData Inventory = new InventorySaveData();
        public PlayerSaveData Player = new PlayerSaveData();

        /// <summary>场景与世界状态（去过哪、拾取、对白、博物等）。</summary>
        public class WorldSaveData
        {
            /// <summary>
            /// 所有加载过的场景（所有该存档角色去过的场景）
            /// </summary>
            public List<string> SavedScenes = new List<string>();
            /// <summary>
            /// 世界中的拾取物数据，数据结构为（拾取物GUID , 是否已被拾取）
            /// </summary>
            public Dictionary<string, bool> PickUps = new Dictionary<string, bool>();
            /// <summary>
            /// 世界中的对白数据，数据结构为（对白GUID , 对白返回值）
            /// </summary>
            public Dictionary<string, short> Dialogs = new Dictionary<string, short>();
            /// <summary>
            /// 博物词条，数据结构为（博物词条ID，是否解锁）
            /// </summary>
            public Dictionary<string, bool> MuseumItem = new Dictionary<string, bool>();
        }

        /// <summary>任务进度（主线 / 支线）。</summary>
        public class QuestSaveData
        {
            /// <summary>
            /// 主线任务
            /// </summary>
            public List<int> MainQuestId = new List<int>();
            /// <summary>
            /// 支线任务
            /// </summary>
            public List<int> SideQuestId = new List<int>();
        }

        /// <summary>背包与金钱。</summary>
        public class InventorySaveData
        {
            /// <summary>
            /// 金钱数据
            /// </summary>
            public int Money = 0;
            /// <summary>
            /// 背包数据，数据结构为 List &lt;（道具ID,道具数目）&gt;
            /// </summary>
            public Dictionary<string, int> BagDatas = new Dictionary<string, int>();
        }

        /// <summary>玩家状态（位置等；按区域扩展时改本类）。</summary>
        public class PlayerSaveData
        {
            public float PosX = 0;
            public float PosY = 0;
            public float PosZ = 0;
        }
    }
}
